* Org Tree Slide

#+begin_src emacs-lisp
  (straight-use-package 'org-tree-slide)

  ;; Simple, writing-focused org presentations
  (with-eval-after-load "org-tree-slide"
    (org-tree-slide-simple-profile)

    ;; Enter/exit presentation: toggle Writeroom and inline images
    (add-hook 'org-tree-slide-play-hook
              (lambda ()
                (when (derived-mode-p 'org-mode)
                  (writeroom-mode 1)
                  (org-display-inline-images))))
    (add-hook 'org-tree-slide-stop-hook
              (lambda ()
                (when (derived-mode-p 'org-mode)
                  (writeroom-mode -1)
                  (org-remove-inline-images)))))

  ;; Start/stop slideshow in Org buffers
  (with-eval-after-load "org"
    (keymap-set org-mode-map "C-c t s" #'org-tree-slide-mode))
#+end_src

* Input method with emacs-rime

~librime~ is required for this feature. Using librime from Squirrel.app.

~emacs-rime~ is the frontend of rime built with emacs input method API.

#+begin_src emacs-lisp
  (straight-use-package 'rime)
  (straight-use-package 'posframe)

  (setq default-input-method "rime"
        ;; librime from Homebrew (provides headers for compilation)
        rime-librime-root "/opt/homebrew"
        ;; Separate user data (avoids database conflicts)
        rime-user-data-dir "~/.config/emacs-rime"
        ;; Use rime-ice config as shared data (schemas, dicts, lua)
        rime-share-data-dir "~/Library/Rime"
        ;; Candidate display
        rime-show-candidate 'posframe)

  (with-eval-after-load "rime"
    ;; Navigation keys when composing
    (dolist (k '("TAB" "C-p" "C-n" "M-v" "C-v"))
      (define-key rime-active-mode-map (kbd k) 'rime-send-keybinding)))

  ;; Auto-disable in code and evil normal mode
  (setq rime-disable-predicates
        '(rime-predicate-prog-in-code-p
          rime-predicate-evil-mode-p))
#+end_src

* Directory environment support with direnv

#+begin_src emacs-lisp
  (straight-use-package 'direnv)

  (setq direnv-always-show-summary nil)

  (autoload 'direnv-mode "direnv" nil t)

  (defun meomacs-direnv-wrapper-function (orig-fun &rest args)
    (if (bound-and-true-p direnv-mode)
        (apply orig-fun args)
      (direnv-mode 1)
      (let ((ret (apply orig-fun args)))
        (direnv-mode -1)
        ret)))

  (defun meomacs-direnv-wrap (command)
    "Wrap a command so direnv starts before the command and stops after the command."
    (advice-add command :around 'meomacs-direnv-wrapper-function))

  (defvar meomacs-direnv-wrap-commands
    '(justl justl-exec-recipe-in-dir))

  (dolist (cmd meomacs-direnv-wrap-commands)
    (meomacs-direnv-wrap cmd))
#+end_src
